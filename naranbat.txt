from tkinter import *
import math
import matplotlib.pyplot as plt
import random

# Funktionen
def ende():
    gui.destroy()
    
def berechnen():
    global dx, x, y
    x=[]
    y=[]
    p_in=p_out=0
    dx=e_dx.get()
    try:
        dx=float(dx)
    except:
        print('Zahl eingeben')
        return
    vx=0
    while vx<=10:
        x.append(vx)
        y.append(math.exp(-2*vx))

        vx+=dx
    for i in range(100000):
        px=random.random()*4+1
        py=random.random()*math.exp(-2)
        if py<=math.exp(-2*px):
            p_in+=1
        else:
            p_out+=1
    fl=4*math.exp(-2)*p_in/(p_in+p_out)
    l_area['text']='Fläche: '+str(fl)
    b_d['state']='normal'
    b_f['state']='normal'
    
def diagramm():
    plt.plot(x,y)
    plt.grid()
    plt.show()

def export():
    filname='C:/users/student08/desktop/test.txt'   
    fileobj=open(filname, 'w')
    for i in range(len(x)):
        hstr=str(round(x[i],3))+', '+str(round(y[i],3))+'\n'
        fileobj.write(hstr)
    fileobj.close()
    
# Formular
gui=Tk()
gui.geometry('300x200')
gui.title('Title')
l_dx=Label(gui, text='dx')
l_area=Label(gui)
e_dx=Entry(gui)
b_e=Button(gui, text='beenden', command=ende, bg='red')
b_b=Button(gui, text='berechnen', command=berechnen)
b_d=Button(gui, text='Diagramm', state='disabled', command=diagramm)
b_f=Button(gui, text='Export', state='disabled', command=export)
l_dx.place(x=30, y=30)
l_area.place(x=30, y=60)
e_dx.place(x=150,y=30)
b_b.place(x=30, y=120)
b_d.place(x=150, y=120)
b_f.place(x=30, y=160)
b_e.place(x=150,y=160)

gui.mainloop()
